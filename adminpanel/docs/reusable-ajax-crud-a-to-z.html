১. মূল ধারণা

প্রতিটি module-এর জন্য আলাদা JavaScript লেখা হবে না। layout.blade.php-এ একটি মাত্র shared CRUD engine থাকবে। User, Section, Category ও Post-এর table/form/button শুধু নির্দিষ্ট data-crud-* attribute ব্যবহার করবে।

Add / Edit / Delete / Status button→Generic JS→Laravel route→Controller + validation→Database→DataTable refresh + Swal
ফলাফল: JavaScript একবার লিখবেন। নতুন module-এ শুধু backend routes/controller এবং Blade table/modal বানাবেন।

৪. layout.blade.php: shared JavaScript engine

নিচের code একবারই layout.blade.php-এ লিখবেন। এটি DataTable, modal form, validation error, save, status, delete ও SweetAlert পরিচালনা করে।

<script>
function initialiseCrudTables() {
  $('[data-crud-table]').each(function () {
    var $table = $(this);
    if (!$.fn.DataTable || $.fn.dataTable.isDataTable($table)) return;
    $table.DataTable({ language: { searchPlaceholder: 'Search...', sSearch: '', lengthMenu: '_MENU_' } });
  });
}

function crudToast(icon, title) {
  Swal.fire({ position: 'top-end', icon: icon, title: title, showConfirmButton: false, timer: 1500 });
}

// বর্তমান page-এর content AJAX দিয়ে নতুন করে নেয়; browser reload হয় না।
function refreshCrudPage(message) {
  window.loadAjaxPage(window.location.href, false);
  setTimeout(function () { crudToast('success', message); }, 250);
}

$(document).on('click', '[data-crud-create]', function () {
  var $button = $(this), $modal = $($button.data('crud-modal'));
  var $form = $modal.find('[data-crud-form]');
  $form[0].reset();
  $form.data({ url: $button.data('store-url'), method: 'POST' });
  $form.find('.js-crud-errors').empty().addClass('d-none');
  $modal.find('[data-crud-title]').text($button.data('create-title') || 'Add Record');
  bootstrap.Modal.getOrCreateInstance($modal[0]).show();
});

$(document).on('click', '[data-crud-edit]', function () {
  var $button = $(this), $modal = $($button.data('crud-modal'));
  var $form = $modal.find('[data-crud-form]');
  $.get($button.data('url')).done(function (response) {
    var record = response.record || response.user || response;
    $form[0].reset();
    $form.data({ url: $button.data('update-url'), method: 'PUT' });
    $.each(record, function (field, value) {
      var $field = $form.find('[name="' + field + '"]');
      if ($field.length && field !== 'password') $field.val(field === 'status' ? (value ? '1' : '0') : (value || ''));
    });
    $form.find('.js-crud-errors').empty().addClass('d-none');
    $modal.find('[data-crud-title]').text('Edit Record');
    bootstrap.Modal.getOrCreateInstance($modal[0]).show();
  });
});

$(document).on('submit', '[data-crud-form]', function (event) {
  event.preventDefault();
  var $form = $(this), data = $form.serializeArray();
  data.push({ name: '_method', value: $form.data('method') });
  $.post($form.data('url'), $.param(data)).done(function (response) {
    bootstrap.Modal.getInstance($form.closest('.modal')[0]).hide();
    refreshCrudPage(response.message || 'Your work has been saved');
  }).fail(function (xhr) {
    if (xhr.status === 422) {
      var messages = $.map(xhr.responseJSON.errors, function (items) { return items.join('<br>'); });
      $form.find('.js-crud-errors').html(messages.join('<br>')).removeClass('d-none');
    }
  });
});

$(document).on('click', '[data-crud-status]', function () {
  $.post($(this).data('url'), { _token: $('[data-crud-form] [name=_token]').first().val(), _method: 'PATCH' })
   .done(function (response) { refreshCrudPage(response.message); });
});

$(document).on('click', '[data-crud-delete]', function () {
  var url = $(this).data('url'), token = $('[data-crud-form] [name=_token]').first().val();
  Swal.fire({ title: 'Are you sure?', icon: 'warning', showCancelButton: true, confirmButtonText: 'Yes, delete it!' })
   .then(function (result) {
     if (result.isConfirmed) $.post(url, { _token: token, _method: 'DELETE' })
       .done(function (response) { refreshCrudPage(response.message); });
   });
});

$(function () { initialiseCrudTables(); });
</script>

৬. Model

class Section extends Model
{
    protected $fillable = ['name', 'status'];

    protected function casts(): array
    {
        return ['status' => 'boolean'];
    }
}

fillable ছাড়া Section::create($data) save হবে না। status boolean cast হওয়ার কারণে JavaScript-এ সহজে Active/Inactive handle করা যায়।

৭. Controller: Section সম্পূর্ণ উদাহরণ

public function section()
{
    $sections = Section::select('id', 'name', 'status')->latest('id')->get();
    $title = 'Section Page';
    return view('admin.sections.section', compact('sections', 'title'));
}

public function store(Request $request)
{
    Section::create($this->validateSection($request));
    return response()->json(['message' => 'Section saved successfully.'], 201);
}

public function show(Section $section)
{
    return response()->json(['record' => $section]);
}

public function update(Request $request, Section $section)
{
    $section->update($this->validateSection($request));
    return response()->json(['message' => 'Section updated successfully.']);
}

public function updateStatus(Section $section)
{
    $section->update(['status' => ! $section->status]);
    return response()->json(['message' => 'Section status updated successfully.']);
}

public function destroy(Section $section)
{
    $section->delete();
    return response()->json(['message' => 'Section deleted successfully.']);
}

private function validateSection(Request $request): array
{
    return $request->validate([
        'name' => ['required', 'string', 'max:255'],
        'status' => ['required', 'boolean'],
    ]);
}
AJAX response সবসময় response()->json() হতে হবে। Validation fail হলে Laravel automatic 422 JSON error পাঠায়, যা generic form code দেখায়।

৯. DataTable A–Z

Table-এ শুধু data-crud-table দিন। Shared code automatic search, pagination ও rows-per-page dropdown চালু করবে।

<table id="sections-table" data-crud-table class="table table-bordered text-nowrap">
  <thead>...</thead>
  <tbody>...</tbody>
</table>

$.fn.dataTable.isDataTable($table) check থাকার কারণে same table দ্বিতীয়বার initialize error হয় না। সফল CRUD-এর পর refreshCrudPage() table-এর নতুন HTML নিয়ে আসে এবং DataTable আবার initialize করে।

১০. নতুন Category বা Post module বানানোর checklist

  1. Migration: table ও প্রয়োজনীয় columns বানান।
  2. Model: $fillable ও দরকার হলে casts() দিন।
  3. Controller: list + store + show + update + updateStatus + destroy method লিখুন।
  4. Routes: উপরের ছয়টি method দিন। সব route admin middleware-এর মধ্যে রাখুন।
  5. Blade partial: table-এ data-crud-table, button/form-এ উপরের attributes দিন।
  6. Sidebar-এ normal Laravel route link দিন; আপনার generic page loader তা reload ছাড়াই open করবে।
যা আর করবেন না: নতুন module-এর জন্য js-category-create, js-post-delete ইত্যাদির আলাদা layout JavaScript লিখবেন না। শুধুই data-crud-* ব্যবহার করুন।
